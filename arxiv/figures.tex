% ===== FIG_PIPELINE =====
\begin{figure}[H]
\centering
\begin{tikzpicture}[
  font=\small,
  box/.style={draw, rounded corners=3pt, align=center, inner sep=5pt,
              minimum height=8mm, text width=#1},
  dec/.style={draw, diamond, aspect=2.2, align=center, inner sep=1pt, font=\small},
  ar/.style={-{Latex[length=2mm]}, thick},
  node distance=6mm,
]

% ---------------- left column: the conventional framing
\node[font=\bfseries] (lh) at (0,0) {Conventional evaluation};
\node[box=26mm, below=5mm of lh] (li) {Embryo image};
\node[box=26mm, below=of li] (lm) {Scoring model\\\footnotesize(CNN, time-lapse)};
\node[box=26mm, below=of lm] (ls) {Scalar score};
\node[box=26mm, below=of ls] (lr) {Ranked embryos};
\node[box=26mm, below=of lr, fill=black!4] (lo) {Measured by\\outcome AUROC};
\draw[ar] (li) -- (lm);
\draw[ar] (lm) -- (ls);
\draw[ar] (ls) -- (lr);
\draw[ar] (lr) -- (lo);

% ---------------- right column: this work
\node[font=\bfseries] (rh) at (6.6,0) {IVF-Bench (this work)};
\node[box=52mm, below=5mm of rh] (ri)
  {Embryo image \textbf{+} patient context\\
   \footnotesize Gardner grade, cycle data, history};
\node[box=52mm, below=of ri] (rm) {Vision-language model};
\node[box=52mm, below=of rm] (ra)
  {Open-ended assessment\\
   \footnotesize morphology, integration, probability, plan};
\node[dec, below=6mm of ra] (rd) {which\\part?};

\node[box=30mm, below left=9mm and -4mm of rd] (rj)
  {Five clinical rubrics\\\footnotesize LLM judge, 1--5};
\node[box=30mm, below right=9mm and -4mm of rd] (rp)
  {Stated probability\\\footnotesize Brier, AUROC};
\node[box=52mm, below=16mm of rd, fill=black!4] (ro)
  {Measured by rubric scores \textbf{and} outcome};

\draw[ar] (ri) -- (rm);
\draw[ar] (rm) -- (ra);
\draw[ar] (ra) -- (rd);
\draw[ar] (rd) -| node[above left, font=\scriptsize, pos=0.25] {reasoning} (rj);
\draw[ar] (rd) -| node[above right, font=\scriptsize, pos=0.25] {prognosis} (rp);
\draw[ar] (rj) |- (ro);
\draw[ar] (rp) |- (ro);
\end{tikzpicture}
\caption{Conventional embryo-selection evaluation against IVF-Bench. Prior work
maps an image to a scalar and is scored only by how well that scalar ranks
outcomes, so a model that reasons well about a patient and a model that does not
are indistinguishable. IVF-Bench supplies the patient context alongside the
image, elicits an open-ended assessment, and scores the reasoning against five
clinical rubrics while separately scoring the stated probability against the
recorded outcome.}
\label{fig:pipeline}
\end{figure}

  xtick=data, x tick label style={font=\small},

  enlarge x limits=0.2,
  ymajorgrids, grid style={black!12},

  nodes near coords, nodes near coords style={font=\tiny, /pgf/number format/fixed,
                                              /pgf/number format/precision=2},

\addplot[fill=black!70, draw=none] coordinates {(GPT-5.4,3.94)(Qwen 397B,3.29)(Gemini Flash,3.24)(Opus 4.6,3.29)};
\addplot[fill=black!18, draw=none] coordinates {(GPT-5.4,1.98)(Qwen 397B,1.98)(Gemini Flash,2.34)(Opus 4.6,3.15)};
\legend{image supplied, image withheld}
\end{axis}
\end{tikzpicture}
\caption{Morphology grounding with the embryo image supplied and withheld, on
identical cases. Removing the image halves GPT-5.4's score and cuts Qwen 397B's
by 40\%, which establishes that the rubric responds to the visual input rather
than to the Gardner grade supplied in text. Opus 4.6 is the exception, losing
only 0.14 points, indicating that its morphology descriptions derive almost
entirely from the annotation rather than the image.}
\label{fig:ablation}
\end{figure}
